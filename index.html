Why use functions?

  • Avoid repeating code
  • Break big problems into small pieces
  • Easy to test and fix

A function is a named block of code. You write it once, then call it many times from anywhere in your program. Functions are the building blocks of every C program.

01

The Simplest Function

#include <stdio.h>

void greet(void) {
    printf("Hello!\n");
}

int main(void) {
    greet();   // call it
    greet();   // call again
    return 0;
}
OUTPUT
Hello!
Hello!
  • void before name → returns nothing
  • void inside () → takes no input
  • greet() → calls the function
02

Function That Returns a Value

#include <stdio.h>

int add(int a, int b) {
    return a + b;
}

int main(void) {
    int result = add(3, 4);
    printf("%d\n", result);
    return 0;
}
OUTPUT
7
  • int before name → returns an integer
  • int a, int b → takes two integers
  • return a + b; → sends answer back

05

Return vs No Return

// Returns value → use result
int square(int x) {
    return x * x;
}

// Returns nothing → just does work
void showSquare(int x) {
    printf("%d\n", x * x);
}

int main(void) {
    int r = square(4);    // store result
    printf("%d\n", r);    // 16

    showSquare(4);        // prints 16 directly
    return 0;
}
OUTPUT
16
16
06

Early Return

#include <stdio.h>

int max(int a, int b) {
    if (a > b) return a;
    return b;
}

int main(void) {
    printf("%d\n", max(10, 20));
    printf("%d\n", max(50, 5));
    return 0;
}
OUTPUT
20
50
Note: return exits the function immediately. Any code after it is skipped.
07

Passing an Array

#include <stdio.h>

void printArray(int a[], int n) {
    for (int i = 0; i < n; i++)
        printf("%d ", a[i]);
    printf("\n");
}

int main(void) {
    int nums[5] = {10, 20, 30, 40, 50};
    printArray(nums, 5);
    return 0;
}
OUTPUT
10 20 30 40 50
Important: You must pass the size n separately. The function has no way to know how long the array is on its own.
08

Pass by Value

In C, functions get copies of values. Changing them inside does NOT change the original.

#include <stdio.h>

void change(int x) {
    x = 100;   // only changes the copy
}

int main(void) {
    int a = 5;
    change(a);
    printf("%d\n", a);   // still 5
    return 0;
}
OUTPUT
5
09

Change Original Using Pointer

To modify the caller's variable, pass its address with &.

#include <stdio.h>

void change(int *x) {
    *x = 100;  // change what x points to
}

int main(void) {
    int a = 5;
    change(&a);          // pass address
    printf("%d\n", a);   // now 100
    return 0;
}
OUTPUT
100
  • &a → address of a
  • int *x → x is a pointer
  • *x = 100 → change the value at that address
10

Swap Two Numbers (Classic Pointer Example)

#include <stdio.h>

void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

int main(void) {
    int x = 1, y = 2;
    swap(&x, &y);
    printf("x = %d, y = %d\n", x, y);
    return 0;
}
OUTPUT
x = 2, y = 1
11

Simple Recursion

A function that calls itself — each call must move toward a base case.

#include <stdio.h>

int factorial(int n) {
    if (n <= 1) return 1;     // base case
    return n * factorial(n - 1);
}

int main(void) {
    printf("%d\n", factorial(5));
    return 0;
}
OUTPUT
120
factorial(5) = 5 * factorial(4)
             = 5 * 4 * factorial(3)
             = 5 * 4 * 3 * factorial(2)
             = 5 * 4 * 3 * 2 * factorial(1)
             = 5 * 4 * 3 * 2 * 1
             = 120
Rule: Always have a base case to stop the recursion, or it runs forever.
12

Function Prototype (Declaration)

If you use a function before defining it, declare it first so the compiler knows it exists.

#include <stdio.h>

int add(int a, int b);  // declaration (prototype)

int main(void) {
    printf("%d\n", add(2, 3));
    return 0;
}

int add(int a, int b) {  // definition
    return a + b;
}
OUTPUT
5
13

Static Local Variable

A static local variable is created once and remembers its value between calls.

#include <stdio.h>

void counter(void) {
    static int n = 0;  // created once, remembered
    n++;
    printf("%d\n", n);
}

int main(void) {
    counter();
    counter();
    counter();
    return 0;
}
OUTPUT
1
2
3

Common Algorithms

15.1 Find Maximum in Array
int findMax(int a[], int n) {
    int m = a[0];
    for (int i = 1; i < n; i++)
        if (a[i] > m) m = a[i];
    return m;
}
OUTPUT — findMax({4,9,2,7,5}, 5)
9
15.2 Sum of Array
int sum(int a[], int n) {
    int total = 0;
    for (int i = 0; i < n; i++)
        total += a[i];
    return total;
}
OUTPUT — sum({1,2,3,4,5}, 5)
15
15.3 Check Even or Odd
int isEven(int n) {
    return n % 2 == 0;
}
OUTPUT — isEven(4), isEven(7)
1 (true)
0 (false)
In C, 1 means true and 0 means false.
15.4 Count Vowels
int countVowels(char s[]) {
    int count = 0;
    for (int i = 0; s[i] != '\0'; i++) {
        char c = s[i];
        if (c=='a'||c=='e'||c=='i'||c=='o'||c=='u'||
            c=='A'||c=='E'||c=='I'||c=='O'||c=='U')
            count++;
    }
    return count;
}
OUTPUT — countVowels("hello world")
3
15.5 Reverse a String In-Place
void reverse(char s[]) {
    int i = 0, j = strlen(s) - 1;
    while (i < j) {
        char t = s[i];
        s[i] = s[j];
        s[j] = t;
        i++; j--;
    }
}
OUTPUT — reverse("hello")
olleh

18

Full Working Example

#include <stdio.h>

int add(int a, int b)       { return a + b; }
int max(int a, int b)       { return (a > b) ? a : b; }

int sumArray(int a[], int n) {
    int s = 0;
    for (int i = 0; i < n; i++) s += a[i];
    return s;
}

void swap(int *a, int *b) {
    int t = *a; *a = *b; *b = t;
}

int factorial(int n) {
    if (n <= 1) return 1;
    return n * factorial(n - 1);
}

int main(void) {
    printf("add:  %d\n", add(3, 4));
    printf("max:  %d\n", max(10, 20));

    int nums[] = {1, 2, 3, 4, 5};
    printf("sum:  %d\n", sumArray(nums, 5));

    int x = 1, y = 2;
    swap(&x, &y);
    printf("swap: x=%d y=%d\n", x, y);

    printf("fact: %d\n", factorial(5));
    return 0;
}
OUTPUT
add: 7
max: 20
sum: 15
swap: x=2 y=1
fact: 120
